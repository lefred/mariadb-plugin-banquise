CMAKE_MINIMUM_REQUIRED(VERSION 3.12)

PROJECT(banquise)

SET(PLUGIN_AUTHOR "lefred")
SET(PLUGIN_VERSION "0.2.1")
SET(PLUGIN_HEX_VERSION "0x000201")
SET(PLUGIN_DESCRIPTION "Standalone Banquise signed-catalog service")
SET(PLUGIN_LICENSE "PLUGIN_LICENSE_GPL")

INCLUDE(CheckIncludeFile)

FIND_PACKAGE(CURL REQUIRED)
FIND_PACKAGE(LibArchive REQUIRED)

SET(BANQUISE_CRYPTO_BACKEND "AUTO" CACHE STRING
  "Crypto backend: AUTO, OPENSSL, or WOLFSSL")
SET_PROPERTY(CACHE BANQUISE_CRYPTO_BACKEND PROPERTY STRINGS AUTO OPENSSL WOLFSSL)

IF(BANQUISE_CRYPTO_BACKEND STREQUAL "AUTO")
  SET(BANQUISE_USE_WOLFSSL OFF)
  IF(EXISTS "${CMAKE_BINARY_DIR}/include/my_config.h")
    FILE(READ "${CMAKE_BINARY_DIR}/include/my_config.h" BANQUISE_MY_CONFIG)
    IF(BANQUISE_MY_CONFIG MATCHES "#define[ \t]+HAVE_WOLFSSL([ \t]|$)")
      SET(BANQUISE_USE_WOLFSSL ON)
    ENDIF()
  ENDIF()
  IF(DEFINED SSL_DEFINES AND SSL_DEFINES MATCHES "HAVE_WOLFSSL")
    SET(BANQUISE_USE_WOLFSSL ON)
  ENDIF()
ELSEIF(BANQUISE_CRYPTO_BACKEND STREQUAL "WOLFSSL")
  SET(BANQUISE_USE_WOLFSSL ON)
ELSEIF(BANQUISE_CRYPTO_BACKEND STREQUAL "OPENSSL")
  SET(BANQUISE_USE_WOLFSSL OFF)
ELSE()
  MESSAGE(FATAL_ERROR "BANQUISE_CRYPTO_BACKEND must be AUTO, OPENSSL, or WOLFSSL")
ENDIF()

IF(BANQUISE_USE_WOLFSSL)
  # MariaDB's bundled wolfSSL requires the same configuration definitions as
  # the server targets (notably WOLFSSL_USER_SETTINGS) before wolfcrypt headers
  # expose the configured algorithms.
  ADD_DEFINITIONS(${SSL_DEFINES})
  IF(EXISTS "${CMAKE_BINARY_DIR}/extra/wolfssl/user_settings.h")
    INCLUDE_DIRECTORIES("${CMAKE_BINARY_DIR}/extra/wolfssl")
  ENDIF()
  SET(BANQUISE_CRYPTO_SOURCES banquise_crypto_wolfssl.cc)
  IF(EXISTS "${CMAKE_SOURCE_DIR}/extra/wolfssl/wolfssl/wolfcrypt/src/blake2b.c")
    # MariaDB's bundled wolfSSL configuration does not build BLAKE2b into
    # libwolfssl, although the catalog signature format requires it.
    ADD_DEFINITIONS(-DHAVE_BLAKE2 -DHAVE_BLAKE2B)
    LIST(APPEND BANQUISE_CRYPTO_SOURCES
      "${CMAKE_SOURCE_DIR}/extra/wolfssl/wolfssl/wolfcrypt/src/blake2b.c")
  ENDIF()
  SET(BANQUISE_WOLFSSL_HAS_ED25519 OFF)
  IF(EXISTS "${CMAKE_BINARY_DIR}/extra/wolfssl/user_settings.h")
    FILE(READ "${CMAKE_BINARY_DIR}/extra/wolfssl/user_settings.h"
      BANQUISE_WOLFSSL_SETTINGS)
    IF(BANQUISE_WOLFSSL_SETTINGS MATCHES "#define[ \t]+HAVE_ED25519([ \t\r\n]|$)")
      SET(BANQUISE_WOLFSSL_HAS_ED25519 ON)
    ENDIF()
  ENDIF()
  SET(BANQUISE_WOLFCRYPT_SRCDIR
    "${CMAKE_SOURCE_DIR}/extra/wolfssl/wolfssl/wolfcrypt/src")
  IF(NOT BANQUISE_WOLFSSL_HAS_ED25519 AND
     EXISTS "${BANQUISE_WOLFCRYPT_SRCDIR}/ed25519.c")
    # MariaDB < 11.6 does not build Ed25519 into its bundled wolfSSL (it was
    # added for PARSEC). Compile it here using the portable C implementation:
    # NO_CURVED25519_X64 avoids needing fe_x25519_asm.S, which MariaDB only
    # assembles when Ed25519 is enabled.
    ADD_DEFINITIONS(-DHAVE_ED25519 -DNO_CURVED25519_X64)
    LIST(APPEND BANQUISE_CRYPTO_SOURCES
      "${BANQUISE_WOLFCRYPT_SRCDIR}/ed25519.c"
      "${BANQUISE_WOLFCRYPT_SRCDIR}/fe_operations.c"
      "${BANQUISE_WOLFCRYPT_SRCDIR}/ge_operations.c")
  ENDIF()
  IF(TARGET wolfssl)
    SET(BANQUISE_CRYPTO_LIBRARY wolfssl)
  ELSEIF(SSL_LIBRARIES)
    SET(BANQUISE_CRYPTO_LIBRARY ${SSL_LIBRARIES})
  ELSE()
    MESSAGE(FATAL_ERROR "WolfSSL selected but no MariaDB wolfSSL library is available")
  ENDIF()
ELSE()
  FIND_PACKAGE(OpenSSL REQUIRED COMPONENTS Crypto)
  SET(BANQUISE_CRYPTO_SOURCES banquise_crypto_openssl.cc)
  SET(BANQUISE_CRYPTO_LIBRARY OpenSSL::Crypto)
ENDIF()

SET(BANQUISE_PLUGIN_SOURCES banquise.cc ${BANQUISE_CRYPTO_SOURCES})
SET(BANQUISE_PLUGIN_LIBRARIES
  CURL::libcurl LibArchive::LibArchive ${BANQUISE_CRYPTO_LIBRARY})

# New MariaDB external plugin infrastructure (MariaDB Foundry).
# QUIET is important so older MariaDB versions can fall back to
# MYSQL_ADD_PLUGIN().
FIND_PACKAGE(mariadb-plugin QUIET)

IF(COMMAND MARIADB_ADD_PLUGIN)
  MESSAGE(STATUS "Building banquise using MARIADB_ADD_PLUGIN")

  MARIADB_ADD_PLUGIN(banquise ${BANQUISE_PLUGIN_SOURCES}
    MODULE_ONLY
    RECOMPILE_FOR_EMBEDDED
    AUTHOR "${PLUGIN_AUTHOR}"
    VERSION "${PLUGIN_VERSION}"
    DESCRIPTION "${PLUGIN_DESCRIPTION}"
    LINK_LIBRARIES ${BANQUISE_PLUGIN_LIBRARIES})
ELSEIF(COMMAND MYSQL_ADD_PLUGIN)
  MESSAGE(STATUS "Building banquise using legacy MYSQL_ADD_PLUGIN")

  MYSQL_ADD_PLUGIN(banquise ${BANQUISE_PLUGIN_SOURCES}
    MODULE_ONLY
    RECOMPILE_FOR_EMBEDDED
    LINK_LIBRARIES ${BANQUISE_PLUGIN_LIBRARIES})

  # MARIADB_ADD_PLUGIN provides these definitions automatically.
  # Legacy MYSQL_ADD_PLUGIN does not, so provide them explicitly.
  IF(TARGET banquise)
    TARGET_COMPILE_DEFINITIONS(banquise PRIVATE
      PLUGIN_AUTHOR="${PLUGIN_AUTHOR}"
      PLUGIN_DESCRIPTION="${PLUGIN_DESCRIPTION}"
      PLUGIN_VERSION="${PLUGIN_VERSION}"
      PLUGIN_HEX_VERSION=${PLUGIN_HEX_VERSION}
      PLUGIN_LICENSE=${PLUGIN_LICENSE})
  ENDIF()
ELSE()
  MESSAGE(FATAL_ERROR
    "Cannot build banquise: neither MARIADB_ADD_PLUGIN nor "
    "MYSQL_ADD_PLUGIN is available.")
ENDIF()

INCLUDE(CPack)
